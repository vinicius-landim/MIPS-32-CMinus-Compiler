int gvar;
int gvet[3];

void processa(int fator, int pvet[]) {
    int localres;
    
    /* Testa LOADARR, LOADVAR e ADD */
    localres = pvet[0] + fator;
    
    /* Testa STOREARR (com o registrador $28) */
    gvet[1] = localres;
}

void main(void) {
    int i;
    
    /* Testa STOREVAR global (com base $0) */
    gvar = 10;
    gvet[0] = 5;
    
    /* Testa passagem de parametros (ARG e ARG_ARR) e CALL */
    processa(gvar, gvet);
}
